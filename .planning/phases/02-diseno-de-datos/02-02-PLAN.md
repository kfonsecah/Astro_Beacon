    ```typescript
      data: T;
      message: string;
      timestamp: Date;
    }
    
    export interface PaginatedResponse<T> {
      data: T[];
      total: number;
      page: number;
      pageSize: number;
      hasMore: boolean;
    }
    
    export type SyncStatus = 'synced' | 'pending' | 'failed';
    
    export interface OfflineQueueItem<T> {
      id: string;
      action: 'create' | 'update' | 'delete';
      data: T;
      queuedAt: Date;
      status: SyncStatus;
      error: string | null;
    }
    ```
