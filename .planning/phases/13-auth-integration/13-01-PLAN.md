### Step 1: Create Auth Service (`src/services/auth.service.ts`)

Create API service functions that call backend endpoints:

```typescript
import api from './api';
import { AUTH_ENDPOINTS } from '@/constants/api';

export interface LoginCredentials {
  email: string;
  password: string;
}

export interface RegisterData {
  email: string;
  password: string;
  name: string;
}

export interface AuthResponse {
  success: boolean;
  data: {
    accessToken: string;
    refreshToken: string;
    user: {
      id: string;
      name: string;
      email: string;
    };
  };
}

export const login = async (credentials: LoginCredentials): Promise<AuthResponse['data']> => {
  const response = await api.post(AUTH_ENDPOINTS.LOGIN, credentials);
  return response; // axios interceptor returns response.data
};

export const register = async (data: RegisterData): Promise<AuthResponse['data']> => {
  const response = await api.post(AUTH_ENDPOINTS.REGISTER, data);
  return response;
};

export const refreshToken = async (refreshToken: string): Promise<{ accessToken: string; refreshToken: string }> => {
  const response = await api.post(AUTH_ENDPOINTS.REFRESH, { refreshToken });
  return response;
};

export const logout = async (): Promise<void> => {
  await api.post(AUTH_ENDPOINTS.LOGOUT);
};
```

### Step 2: Create Auth Hooks (`src/hooks/useAuth.ts`)

Create TanStack Query mutations for auth operations:

```typescript
import { useMutation } from '@tanstack/react-query';
import { useRouter } from 'expo-router';
import { useAuthStore } from '@/stores/auth.store';
import { login, register, refreshToken, logout } from '@/services/auth.service';

export function useLogin() {
  const router = useRouter();
  const setAuth = useAuthStore((state) => state.setAuth);

  return useMutation({
    mutationFn: login,
    onSuccess: async (data) => {
      // Store tokens in Zustand (auto-persisted via persist middleware)
      setAuth(data.accessToken, data.refreshToken, data.user);
      // Navigate to home, replace to prevent back to login
      router.replace('/(app)/(tabs)/home');
    },
    onError: (error: any) => {
      // Error handled by axios interceptor + form displays field errors
      console.warn('[useLogin] Failed:', error.message);
    },
  });
}

export function useRegister() {
  const router = useRouter();
  const setAuth = useAuthStore((state) => state.setAuth);

  return useMutation({
    mutationFn: register,
    onSuccess: async (data) => {
      // Backend auto-logs in after register (returns tokens)
      setAuth(data.accessToken, data.refreshToken, data.user);
      // Navigate to home, replace to prevent back to login
      router.replace('/(app)/(tabs)/home');
    },
    onError: (error: any) => {
      console.warn('[useRegister] Failed:', error.message);
    },
  });
}

export function useRefreshToken() {
  return useMutation({
    mutationFn: ({ refreshToken }: { refreshToken: string }) => refreshToken(refreshToken),
  });
}

export function useLogout() {
  const clearAuth = useAuthStore((state) => state.clearAuth);

  return useMutation({
    mutationFn: logout,
    onSettled: async () => {
      // Clear store regardless of API success (token might be expired)
      clearAuth();
    },
  });
}
```

### Step 3: Update Auth Store (`src/stores/auth.store.ts`)

Replace mock implementation with real API integration and Zustand persist:

```typescript
import { create } from 'zustand';
import { persist, createJSONStorage } from 'zustand/middleware';
import * as SecureStore from 'expo-secure-store';

interface AuthUser {
  id: string;
  name: string;
  email: string;
}

interface AuthState {
  accessToken: string | null;
  refreshToken: string | null;
  user: AuthUser | null;
  isAuthenticated: boolean;
  isLoading: boolean;
  
  // Actions
  setAuth: (accessToken: string, refreshToken: string, user: AuthUser) => void;
  clearAuth: () => void;
  setLoading: (loading: boolean) => void;
}

// SecureStore adapter for Zustand persist
const secureStoreStorage = {
  getItem: async (name: string): Promise<string | null> => {
    return await SecureStore.getItemAsync(name);
  },
  setItem: async (name: string, value: string): Promise<void> => {
    await SecureStore.setItemAsync(name, value);
  },
  removeItem: async (name: string): Promise<void> => {
    await SecureStore.deleteItemAsync(name);
  },
};

export const useAuthStore = create<AuthState>()(
  persist(
    (set) => ({
      accessToken: null,
      refreshToken: null,
      user: null,
      isAuthenticated: false,
      isLoading: true, // Start as loading (checking SecureStore)

      setAuth: (accessToken, refreshToken, user) => {
        set({
          accessToken,
          refreshToken,
          user,
          isAuthenticated: true,
          isLoading: false,
        });
      },

      clearAuth: () => {
        set({
          accessToken: null,
          refreshToken: null,
          user: null,
          isAuthenticated: false,
          isLoading: false,
        });
      },

      setLoading: (loading) => {
        set({ isLoading: loading });
      },
    }),
    {
      name: 'auth-storage', // Not used directly, we specify storage per key
      storage: createJSONStorage(() => secureStoreStorage),
      partialize: (state) => ({
        accessToken: state.accessToken,
        refreshToken: state.refreshToken,
        user: state.user,
      }),
      onRehydrateStorage: () => (state) => {
        if (state?.accessToken) {
          state.isAuthenticated = true;
        }
        state?.setLoading(false);
      },
    }
  )
);

// Alias para componentes
export const useAuth = useAuthStore;
```

**Verification:**
- [ ] Zustand persist middleware added (API-06)
- [ ] Storage keys: `accessToken`, `refreshToken`, `user` (CONTEXT D-15, D-16)
- [ ] Old key `'auth_token'` migrated to new separate keys (CONTEXT D-17)
- [ ] `isLoading` starts as `true`, set to `false` after rehydration
- [ ] `setAuth()` and `clearAuth()` actions defined

### Step 4: Update Register Screen (`app/(auth)/register.tsx`)

Consume `useRegister` mutation with loading/error states:

```typescript
import { useState } from 'react';
import { View, Text, TextInput, TouchableOpacity, ActivityIndicator } from 'react-native';
import { useRegister } from '@/hooks/useAuth';
import { useTheme } from '@/theme/useTheme';
import { spacing } from '@/constants/spacing';

export default function RegisterScreen() {
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [name, setName] = useState('');
  const [errors, setErrors] = useState<Record<string, string>>({});
  
  const { mutate: register, isPending, error } = useRegister();
  const theme = useTheme();
  const { colors } = theme;

  const handleRegister = () => {
    // Basic validation
    const newErrors: Record<string, string> = {};
    if (!email) newErrors.email = 'Email is required';
    if (!password) newErrors.password = 'Password is required';
    if (!name) newErrors.name = 'Name is required';
    
    if (Object.keys(newErrors).length > 0) {
      setErrors(newErrors);
      return;
    }
    
    setErrors({});
    register({ email, password, name });
  };

  return (
    <View style={{ flex: 1, padding: spacing.lg, backgroundColor: colors.background }}>
      <Text style={{ color: colors.text, fontSize: 24, marginBottom: spacing.md }}>
        Register
      </Text>
      
      <TextInput
        placeholder="Name"
        value={name}
        onChangeText={setName}
        style={{ borderBottomWidth: 1, borderBottomColor: colors.textMuted, marginBottom: spacing.sm }}
      />
      {errors.name && <Text style={{ color: colors.danger }}>{errors.name}</Text>}
      
      <TextInput
        placeholder="Email"
        value={email}
        onChangeText={setEmail}
        keyboardType="email-address"
        autoCapitalize="none"
        style={{ borderBottomWidth: 1, borderBottomColor: colors.textMuted, marginBottom: spacing.sm }}
      />
      {errors.email && <Text style={{ color: colors.danger }}>{errors.email}</Text>}
      
      <TextInput
        placeholder="Password"
        value={password}
        onChangeText={setPassword}
        secureTextEntry
        style={{ borderBottomWidth: 1, borderBottomColor: colors.textMuted, marginBottom: spacing.md }}
      />
      {errors.password && <Text style={{ color: colors.danger }}>{errors.password}</Text>}
      
      {error && (
        <Text style={{ color: colors.danger, marginBottom: spacing.sm }}>
          {error.message || 'Registration failed'}
        </Text>
      )}
      
      <TouchableOpacity
        onPress={handleRegister}
        disabled={isPending}
        style={{
          backgroundColor: isPending ? colors.textMuted : colors.primary,
          padding: spacing.md,
          borderRadius: 0, // HUD aesthetic: zero border-radius
          alignItems: 'center',
        }}
      >
        {isPending ? (
          <ActivityIndicator color={colors.background} />
        ) : (
          <Text style={{ color: colors.background, fontWeight: 'bold' }}>Register</Text>
        )}
      </TouchableOpacity>
    </View>
  );
}
```
