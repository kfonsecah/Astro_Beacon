### Step 2: Update API Client (`src/services/api.ts`)

Add refresh token rotation with queue for parallel 401s:

```typescript
import axios, { AxiosError, AxiosInstance } from 'axios';
import { useAuthStore } from '@/stores/auth.store';

const API_BASE_URL = process.env.EXPO_PUBLIC_API_URL || (
  __DEV__ ? 'http://localhost:3000/api/v1' : 'https://api.astrobeacon.com/api/v1'
);

const API_TIMEOUT = 15000;

const api: AxiosInstance = axios.create({
  baseURL: API_BASE_URL,
  timeout: API_TIMEOUT,
  headers: {
    'Content-Type': 'application/json',
  },
});

// Refresh token queue state
let isRefreshing = false;
let failedQueue: Array<{
  resolve: (token: string) => void;
  reject: (error: any) => void;
}> = [];

const processQueue = (error: any, token: string | null = null) => {
  failedQueue.forEach(({ resolve, reject }) => {
    if (error) {
      reject(error);
    } else {
      resolve(token!);
    }
  });
  failedQueue = [];
};

// Request interceptor: attach JWT token
api.interceptors.request.use(
  (config) => {
    const token = useAuthStore.getState().accessToken;
    if (token) {
      config.headers.Authorization = `Bearer ${token}`;
    }
    return config;
  },
  (error) => Promise.reject(error)
);

// Response interceptor: handle errors + refresh token
api.interceptors.response.use(
  (response) => response.data, // Unwrap response envelope
  async (error: AxiosError) => {
    const originalRequest = error.config;

    // Network error (no response)
    if (!error.response) {
      if (__DEV__) console.warn('[API] Network error');
      return Promise.reject({
        message: 'Network error. Please check your connection.',
        status: 0,
      });
    }

    const { status, data } = error.response;

    // Handle 401: Token expired
    if (status === 401 && !originalRequest._retry) {
      originalRequest._retry = true;

      const refreshToken = useAuthStore.getState().refreshToken;

      // No refresh token = logout immediately
      if (!refreshToken) {
        if (__DEV__) console.warn('[API] No refresh token, logging out');
        await useAuthStore.getState().clearAuth();
        return Promise.reject({ message: 'Session expired', status: 401 });
      }

      // If already refreshing, queue this request
      if (isRefreshing) {
        return new Promise((resolve, reject) => {
          failedQueue.push({ resolve, reject });
        })
          .then((token) => {
            originalRequest.headers.Authorization = `Bearer ${token}`;
            return api(originalRequest);
          })
          .catch((err) => Promise.reject(err));
      }

      // Start refresh
      isRefreshing = true;

      try {
        // Call refresh endpoint
        const response = await axios.post(
          `${API_BASE_URL}/auth/refresh`,
          { refreshToken },
          { headers: { 'Content-Type': 'application/json' } }
        );

        const { accessToken, refreshToken: newRefreshToken } = response.data.data;

        // Update store with new tokens
        useAuthStore.getState().setAuth(accessToken, newRefreshToken, useAuthStore.getState().user!);

        // Process queued requests
        processQueue(null, accessToken);

        // Retry original request
        originalRequest.headers.Authorization = `Bearer ${accessToken}`;
        return api(originalRequest);
      } catch (refreshError: any) {
        // Refresh failed = logout
        if (__DEV__) console.warn('[API] Refresh failed, logging out');
        await useAuthStore.getState().clearAuth();
        processQueue(refreshError, null);
        return Promise.reject({ message: 'Session expired', status: 401 });
      } finally {
        isRefreshing = false;
      }
    }

    // Handle 403: Forbidden
    if (status === 403) {
      if (__DEV__) console.warn('[API] Forbidden');
      return Promise.reject({
        message: (data as any)?.message || 'You don\'t have permission',
        status: 403,
      });
    }

    // Handle 400: Validation error (return for form handling)
    if (status === 400) {
      return Promise.reject({
        message: (data as any)?.message || 'Invalid request',
        errors: (data as any)?.errors, // Field-specific errors
        status: 400,
      });
    }

    // Handle 500: Server error
    if (status >= 500) {
      if (__DEV__) console.warn('[API] Server error');
      return Promise.reject({
        message: 'Server error. Please try again later.',
        status,
      });
    }

    // Default: return error data
    return Promise.reject({
      message: (data as any)?.message || 'An error occurred',
      status,
    });
  }
);

export { api, API_BASE_URL };
```

**Verification:**
- [ ] Refresh token queue implemented (CONTEXT D-01, D-02)
- [ ] `isRefreshing` flag prevents multiple refresh calls (parallel 401s handled)
- [ ] `processQueue()` retries queued requests with new token
- [ ] 401 → calls `/auth/refresh` → retries original request (AUTH-07)
- [ ] 403/400/500 errors handled with user-friendly messages (API-03)
- [ ] Network errors (no response) detected and handled
- [ ] `useAuthStore.getState().clearAuth()` called on refresh failure (CONTEXT D-03)
- [ ] Response unwrapped: `response.data` returned (per Phase 8 decision D-04: response envelope)

### Step 3: Update Root Layout (`app/_layout.tsx`)

Add Stack.Protected for (app) group:

```typescript
import { Stack } from 'expo-router';
import { QueryClient, QueryClientProvider } from '@tanstack/react-query';
import { useAuth } from '@/stores/auth.store';

// Create QueryClient (Phase 14 will move this to a provider)
const queryClient = new QueryClient();

export default function RootLayout() {
  const { isAuthenticated, isLoading } = useAuth();

  // Show loading while checking auth state
  if (isLoading) {
    return (
      <View style={{ flex: 1, justifyContent: 'center', alignItems: 'center' }}>
        <ActivityIndicator size="large" color={colors.primary} />
      </View>
    );
  }

  return (
    <QueryClientProvider client={queryClient}>
      <Stack>
        {/* Auth group - no protection */}
        <Stack.Screen name="(auth)" options={{ headerShown: false }} />
        
        {/* App group - protected */}
        <Stack.Protected group="(app)" guard={() => isAuthenticated}>
          <Stack.Screen name="(app)" options={{ headerShown: false }} />
        </Stack.Protected>
      </Stack>
    </QueryClientProvider>
  );
}
```

**Note:** This is a simplified version. Phase 14 will properly set up TanStack Query provider. For now, we add the Protected wrapper.
