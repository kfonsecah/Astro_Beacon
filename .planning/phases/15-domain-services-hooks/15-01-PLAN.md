## Backend Service Methods Reference

### auth.service.ts (Phase 9)
- register(input: RegisterInput) → { user, accessToken, refreshToken }
- login(input: LoginInput) → { user, accessToken, refreshToken }
- refresh(refreshToken: string) → { accessToken }
- logout(userId: string, refreshToken?: string) → { message }

### astronaut.service.ts
- createOrUpdate(userId, input) → IAstronaut
- findByUser(userId) → IAstronaut | null
- update(userId, astronautId, input) → IAstronaut
- getDashboard(userId) → { resourcesCount, activeTrips, speciesDiscovered }

### resource.service.ts
- create(userId, input) → IResource
- findAll(userId, page, limit) → { resources, pagination }
- findOne(userId, resourceId) → IResource
- update(userId, resourceId, input) → IResource
- delete(userId, resourceId) → void
- recordMovement(userId, resourceId, input) → IResource
- getAlerts(userId) → ResourceAlert[]

### logbook.service.ts
- create(userId, input, speciesId?) → ILogbookEntry
- findAll(userId, page, limit, speciesId?) → { entries, pagination }
- findOne(userId, entryId) → ILogbookEntry
- update(userId, entryId, input) → ILogbookEntry
- delete(userId, entryId) → void

### species.service.ts
- create(userId, input) → ISpecies
- findAll(userId, page, limit) → { species, pagination }
- findOne(userId, speciesId) → ISpecies

### trip.service.ts
- create(userId, input) → ITrip
- findAll(userId, page, limit, status?) → { trips, pagination }
- findOne(userId, tripId) → ITrip
- update(userId, tripId, input) → ITrip
- start(userId, tripId, input?) → ITrip
- complete(userId, tripId, input?) → ITrip
- abort(userId, tripId, input?) → ITrip
- delete(userId, tripId) → void

### supply.service.ts
- create(userId, input) → ISupply
- findAll(userId, page, limit, status?) → { supplies, pagination }
- findOne(userId, supplyId) → ISupply
- collect(userId, supplyId, input?) → ISupply
- findNearby(userId, lat, lng, radius, status?) → ISupply[]
